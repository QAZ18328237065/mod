# 不定向大圣战信仰资格(任意领袖教义版)
# 复制自原版 suitable_faith_for_undirected_ghw_trigger(00_great_holy_war_triggers.txt:8),
# 差异:原版要求 doctrine_spiritual_head(仅精神领袖),本触发器放宽为精神/世俗领袖均可。
# 触发它的 zzz_crusades_ghw.0001 事件额外硬性要求 doctrine_temporal_head,
# 因此精神领袖信仰仍走原版 great_holy_war.0001,世俗领袖信仰走本 mod 事件,二者互不重复。
zzz_suitable_faith_for_undirected_ghw_any_head_trigger = {
	OR = {
		has_doctrine_parameter = great_holy_wars_active
		AND = {
			has_doctrine_parameter = great_holy_wars_active_if_reformed
			NOT = { has_doctrine_parameter = unreformed }
		}
		has_doctrine = divine_destiny_doctrine
	}
	OR = {
		has_doctrine = doctrine_spiritual_head
		has_doctrine = doctrine_temporal_head
	}
	NOR = {
		has_doctrine_parameter = unreformed
		AND = {
			has_doctrine = tenet_dharmic_pacifism
			NOT = { has_doctrine = divine_destiny_doctrine }
		}
		has_doctrine = tenet_pacifism
	}
	fervor >= 75
	save_temporary_scope_value_as = {
		name = faith_strength
		value = faith_military_strength
	}
	save_temporary_scope_as = faith
	any_kingdom = {
		valid_ghw_target_trigger = { FAITH = scope:faith }
	}
}
