namespace = zzz_crusades_ghw

# 世俗领袖信仰的大圣战发动器
# 原版 great_holy_war.0001(精神领袖信仰的 AI 发动入口)硬编码了 has_doctrine = doctrine_spiritual_head,
# 事件 ID 无法重定义,因此复制其通用发动流程为本事件,面向世俗领袖(如哈里发)信仰。
# 触发条件与原版 0001 互斥(本事件硬性要求 doctrine_temporal_head),不会重复发动。
# 发动后的完整流程(准备阶段/宣誓/捐金/受益人/获胜封王)全部由原版引擎驱动。
zzz_crusades_ghw.0001 = {
	hidden = yes
	scope = faith

	trigger = {
		has_doctrine = doctrine_temporal_head # 世俗领袖专用,精神领袖仍走原版 0001
		save_temporary_scope_as = the_faith
		exists = religious_head
		NOR = {
			exists = great_holy_war
			exists = var:variable_ghw_cooldown
			religion = { exists = var:variable_first_ghw_cooldown } # 解锁后等待约 4 个月再发动
		}
		religious_head = { is_imprisoned = no }
		religion = { exists = var:variable_ghw_unlocked } # 本 mod 开局为全部宗教族解锁
		zzz_suitable_faith_for_undirected_ghw_any_head_trigger = yes # 信条/狂热/有效目标
	}

	immediate = {
		save_scope_as = the_faith

		# 确定教主与目标
		religious_head = { save_scope_as = ghw_sponsor }
		determine_ghw_target_effect = { FAITH = scope:the_faith }

		# 开启准备阶段(倒计时约 1-1.5 年)
		start_great_holy_war = {
			target_character = scope:target_character
			target_title = scope:target_kingdom
			delay = { 365 545 }
		}
		scope:target_character.faith = { save_scope_as = target_faith } # 用于防守方宣誓

		great_holy_war = {
			set_war_declarer = scope:ghw_sponsor
			if = {
				limit = { is_directed_ghw = no }
				undirected_ghw_recipient_selection_effect = yes
			}
		}
		# 教主把一半财富(上限 5000)投入战争金库
		if = {
			limit = { scope:ghw_sponsor.gold > 0 }
			great_holy_war = {
				change_war_chest_gold = {
					value = scope:ghw_sponsor.gold
					divide = 2
					max = 5000
				}
			}
			scope:ghw_sponsor = {
				remove_short_term_gold = {
					value = gold
					divide = 2
					max = 5000
				}
			}
		}
		great_holy_war = {
			pledge_attacker = scope:ghw_sponsor
			set_variable = {
				name = redirect_cost
				value = 500
			}
		}
		# 无候选受益人时生成后备受益人,保证宣战界面完整
		if = {
			limit = {
				NOT = { exists = scope:the_faith.great_holy_war.ghw_title_recipient }
				exists = scope:ghw_sponsor
				exists = scope:the_faith.great_holy_war
				scope:ghw_sponsor = {
					any_courtier = {
						is_adult = yes
						is_ai = yes
						NOR = {
							has_trait = excommunicated
							has_trait = incapable
							has_trait = devoted
						}
						faith = scope:the_faith
					}
				}
			}
			scope:ghw_sponsor = {
				random_courtier = {
					alternative_limit = {
						scope:the_faith = { has_doctrine = doctrine_gender_male_dominated }
						is_male = yes
						is_adult = yes
						is_ai = yes
						NOR = {
							has_trait = excommunicated
							has_trait = incapable
							has_trait = devoted
						}
						faith = scope:the_faith
					}
					alternative_limit = {
						scope:the_faith = { has_doctrine = doctrine_gender_female_dominated }
						is_female = yes
						is_adult = yes
						is_ai = yes
						NOR = {
							has_trait = excommunicated
							has_trait = incapable
							has_trait = devoted
						}
						faith = scope:the_faith
					}
					limit = {
						is_adult = yes
						is_ai = yes
						NOR = {
							has_trait = excommunicated
							has_trait = incapable
							has_trait = devoted
						}
						faith = scope:the_faith
					}
					save_scope_as = fallback_recipient
					add_character_flag = flag_is_ghw_fallback_recipient
				}
			}
		}
		else_if = { # 教主无合适朝臣则生成一个
			limit = {
				NOT = { exists = scope:the_faith.great_holy_war.ghw_title_recipient }
			}
			create_character = {
				location = scope:ghw_sponsor.capital_province
				template = new_commander_character
				faith = scope:the_faith
				culture = scope:ghw_sponsor.culture
				save_scope_as = fallback_recipient
				gender_female_chance = {
					if = {
						limit = {
							scope:the_faith = { has_doctrine = doctrine_gender_male_dominated }
						}
						add = 0
					}
					else_if = {
						limit = {
							scope:the_faith = { has_doctrine = doctrine_gender_female_dominated }
						}
						add = 100
					}
					else = {
						add = 50
					}
				}
			}
			scope:fallback_recipient = { add_character_flag = flag_is_ghw_fallback_recipient }
		}
		great_holy_war = {
			set_variable = {
				name = var_fallback_recipient
				value = scope:fallback_recipient
			}
		}

		# 通知进攻方玩家准备阶段开始(原版事件,各宗教均有通用文案)
		every_player = {
			limit = {
				faith = scope:the_faith
				highest_held_title_tier >= tier_county
				this != scope:ghw_sponsor
			}
			trigger_event = great_holy_war.0003
		}
		# 通知防守方玩家
		every_player = {
			limit = {
				OR = {
					faith = scope:target_faith # 目标王国信仰所属
					any_liege_or_above = {
						scope:the_faith.great_holy_war = { has_pledged_defender = prev }
					}
					AND = {
						faith != scope:the_faith
						scope:the_faith.great_holy_war = { has_pledged_defender = prev }
					}
				}
			}
			trigger_event = great_holy_war.0050
		}

		# 攻击方 AI 宣誓(有概率在开战后才加入)
		every_faith_ruler = {
			limit = {
				is_ai = yes
				is_landed = yes
				highest_held_title_tier >= tier_county
				NOR = {
					this = scope:ghw_sponsor
					root.great_holy_war = { has_pledged_attacker = prev }
					target_is_liege_or_above = scope:target_character
				}
			}
			trigger_event = {
				id = great_holy_war.0020
				days = { 50 450 }
			}
		}
		# 骑士团必定加入
		every_faith_holy_order = {
			trigger_event = {
				id = great_holy_war.0090
				days = { 30 60 }
			}
		}

		# 防守方 AI 宣誓
		scope:target_character = {
			religion = {
				every_faith = {
					every_faith_ruler = {
						limit = {
							is_ai = yes
							is_landed = yes
							highest_held_title_tier >= tier_county
							NOT = {
								root.great_holy_war = { has_pledged_defender = prev }
							}
						}
						trigger_event = {
							id = great_holy_war.0021
							days = { 25 150 }
						}
					}
				}
			}
			faith = {
				every_faith_holy_order = {
					trigger_event = {
						id = great_holy_war.0091
						days = { 30 60 }
					}
				}
			}
		}
	}
}
