# 朝贡讨伐 CB:宗主对断贡叛逃者的专属宣战理由
# 合法性核心:目标必须在宗主报复列表(subjugated_tributary_rebellion_retaliation_target_list)中
# 战争结果:胜利 -> 叛逃者重新成为朝贡者(与原版 make_tributary_cb 相同)
subjugated_tributary_rebellion_retaliation_cb = {
	icon = make_tributary_cb
	group = subjugation
	ai_only_against_neighbors = yes

	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	allowed_for_character = {
		can_have_tributaries_trigger = yes
		is_landed = yes
	}

	allowed_against_character = {
		# 只能打记在账上的叛逃者(报复列表,支持多个并存)
		scope:attacker = {
			is_target_in_variable_list = {
				name = subjugated_tributary_rebellion_retaliation_target_list
				target = scope:defender
			}
		}
		scope:defender = {
			is_independent_ruler = yes
			is_landed = yes
			NOT = { government_has_flag = cannot_be_vassal_or_liege }
		}
	}

	allowed_against_character_display_regardless = {
		custom_tooltip = {
			text = must_neighbor_or_neighbor_tributary_tt
			scope:attacker = {
				OR = {
					character_is_realm_neighbor = scope:defender
					any_neighboring_realm_with_tributaries_owner = {
						this = scope:defender
					}
				}
			}
		}
	}

	should_invalidate = {
		OR = {
			scope:attacker = {
				OR = {
					is_independent_ruler = no
					is_landed = no
				}
			}
			scope:defender = {
				OR = {
					is_independent_ruler = no
					is_landed = no
				}
			}
		}
	}

	cost = {
		prestige = {
			value = {
				scope:defender = {
					add = {
						add = realm_size
						multiply = 25
						if = {
							limit = { primary_title.tier = tier_county }
							max = 100
						}
						else_if = {
							limit = { primary_title.tier = tier_duchy }
							max = 200
						}
						else = {
							max = 300
						}
						desc = CB_SIZE_OF_TARGET_REALM_COST
					}
				}
			}
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_invalidated_desc = msg_make_tributary_war_invalidated_message

	on_invalidated = {}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = subjugated_tributary_rebellion_retaliation_cb_victory_desc_attacker
			}
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = subjugated_tributary_rebellion_retaliation_cb_victory_desc_defender
			}
			desc = subjugated_tributary_rebellion_retaliation_cb_victory_desc
		}
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }

		# 收服成功:从报复列表移除该目标
		scope:attacker = {
			remove_list_variable = {
				name = subjugated_tributary_rebellion_retaliation_target_list
				target = scope:defender
			}
		}

		# 核心:打赢 = 重新被征服(被征服朝贡组,而非普通朝贡)
		scope:defender = {
			start_tributary = {
				contract_group = tributary_subjugated
				suzerain = scope:attacker
			}
			# 宗主快照直接建档,避免下次季检误走"首次建档"逻辑
			set_variable = {
				name = subjugated_tributary_rebellion_subjugation_suzerain
				value = scope:attacker
			}
			# 重新被征服:臣服期重启(10 年,期间无力反抗)
			set_variable = {
				name = subjugated_tributary_rebellion_subjugation_calm
				value = yes
				days = 3650
			}
			# 怨气:重新被征服的人心怀怨恨,但臣服期内无力反抗
			add_opinion = {
				modifier = subjugated_tributary_rebellion_resent_new_suzerain_opinion
				target = scope:attacker
				opinion = -30
			}
		}

		if = {
			limit = {
				scope:defender = {
					is_confederation_member = yes
				}
			}
			scope:defender.confederation = {
				remove_confederation_member = scope:defender
			}
		}

		scope:attacker = {
			add_prestige_experience = {
				value = medium_prestige_value
			}
		}

		scope:defender = {
			add_prestige = {
				value = medium_prestige_value
				multiply = -1.0
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = major_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_attacker_victory_effect = yes

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = subjugated_tributary_rebellion_retaliation_cb_white_peace_desc_defender
			}
			desc = subjugated_tributary_rebellion_retaliation_cb_white_peace_desc
		}
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }

		scope:defender = { accolade_defender_war_end_glory_gain_low_effect = yes }

		scope:attacker = {
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
			stress_impact = {
				ambitious = medium_stress_impact_gain
				arrogant = medium_stress_impact_gain
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_white_peace_effect = yes

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = subjugated_tributary_rebellion_retaliation_cb_white_peace_desc_defender
			}
			triggered_desc = {
				trigger = {
					scope:attacker = {
						is_local_player = yes
						has_targeting_faction = yes
					}
				}
				desc = subjugated_tributary_rebellion_retaliation_cb_defeat_desc_attacker
			}
			desc = subjugated_tributary_rebellion_retaliation_cb_white_peace_desc
		}
	}

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }

		scope:defender = {
			accolade_defender_war_end_glory_gain_high_effect = yes
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = medium_prestige_value
			}
		}

		scope:attacker = {
			add_prestige = {
				value = major_prestige_value
				multiply = -1.0
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = major_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		scope:attacker = {
		save_temporary_scope_as = loser
	}

	add_truce_attacker_defeat_effect = yes

		on_lost_aggression_war_discontent_loss = yes

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_primary_attacker_death = invalidate
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "SUBJUGATED_TRIBUTARY_REBELLION_RETALIATION_WAR_NAME"
	war_name_base = "SUBJUGATED_TRIBUTARY_REBELLION_RETALIATION_WAR_NAME_BASE"
	cb_name = "SUBJUGATED_TRIBUTARY_REBELLION_RETALIATION_CB_NAME"
	interface_priority = 60

	ticking_war_score_targets_entire_realm = yes
	max_defender_score_from_battles = 150
	max_attacker_score_from_battles = 150
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
	attacker_ticking_warscore = 0
	attacker_wargoal_percentage = 0.8

	max_ai_diplo_distance_to_title = 500

	ai_score_mult = {
		value = 1
		if = {
			limit = {
				scope:defender.primary_title.tier >= scope:attacker.primary_title.tier
			}
			multiply = 0.1
		}
	}

	ai_score = {
		value = {
			value = 50
			scope:defender = {
				every_sub_realm_county = {
					add = 50
				}
			}
			scope:attacker = {
				if = {
					limit = {
						NOT = {
							government_has_flag = government_is_nomadic
						}
					}
					multiply = {
						value = 0.2
						if = {
							limit = {
								scope:defender.monthly_character_income >= 3
								scope:defender.monthly_character_income >= {
									add = scope:attacker.monthly_character_income
									multiply = 1.25
								}
							}
							add = 0.3
						}
					}
				}
			}
		}
	}
}
