# 被征服朝贡者季度检查(scope = 朝贡者自身)
# 1. 宗主快照建档/换主检测(反叛期标记 + 记录)
# 2. 反抗期内 AI 每年一次反抗掷骰(权重见 zzz_tribsub_tributary_rebellion_chance_value)
zzz_tribsub_subjugated_tributary_check_effect = {
	# ---------- 宗主快照比对 ----------
	if = {
		limit = {
			NOT = { exists = var:tribsub_subjugation_suzerain }
		}
		# 首次建档:记录当前宗主,不打标记
		set_variable = { name = tribsub_subjugation_suzerain value = suzerain }
		# 被征服朝贡者死亡后由继承人继承朝贡关系:通知玩家
		if = {
			limit = {
				is_local_player = yes
			}
			trigger_event = { id = zzz_tribsub_subjugation.0003 }
		}
	}
	else_if = {
		limit = {
			exists = suzerain
			NOT = { var:tribsub_subjugation_suzerain ?= suzerain }
		}
		# ---------- 换宗主了:记录反抗 ----------
		set_variable = { name = tribsub_subjugation_suzerain value = suzerain }
		set_variable = {
			name = tribsub_resisting_new_suzerain
			value = yes
			days = 730
		}
		# 藩属关系与好感痕迹(永久 -30 怨愤,作为"反抗记录"留在朝贡者身上)
		add_subject_standing = -10
		add_opinion = {
			modifier = tribsub_resent_new_suzerain_opinion
			target = suzerain
			opinion = -30
		}
		# 玩家自己是被征服朝贡者:宗主更替通知
		if = {
			limit = {
				is_local_player = yes
			}
			trigger_event = { id = zzz_tribsub_subjugation.0002 }
		}
	}

	# ---------- 反抗期 AI 宣战判定(每年一次) ----------
	if = {
		limit = {
			has_variable = tribsub_resisting_new_suzerain
			is_ai = yes
			NOT = { exists = var:tribsub_rebellion_roll_cooldown }
		}
		set_variable = {
			name = tribsub_rebellion_roll_cooldown
			value = yes
			days = 365
		}
			random = {
			chance = zzz_tribsub_tributary_rebellion_chance_value
			# 掷中:有底气就宣独立战争,否则直接断绝朝贡
			if = {
				limit = {
					is_at_war = no
				}
				ai_start_best_war = {
					cb = { tributary_independence_war }
					is_valid = {
						scope:target_character = {
							is_landed = yes
						}
					}
				}
				# 宣战未成(打不过/休战期等,此刻仍无战事),转为直接断绝朝贡
				# 注意:若 ai_start_best_war 宣战成功,朝贡者已 is_at_war = yes,
				# 此时绝不能 end_tributary——否则独立战争瞬间失去战争目标而失效
				if = {
					limit = {
						is_tributary = yes
						is_at_war = no
					}
					# 宗主记账:此朝贡者叛逃,并弹出三选一应对事件
					# (不设冷却:若多朝贡者先后断贡,逐个记账、逐个弹事件,避免账目丢失)
					# 触发前防御:叛逃者必须仍独立,否则不弹事件(交给季检清理)
					suzerain = {
						set_variable = {
							name = tribsub_wayward_tributary
							value = prev
						}
						if = {
							limit = {
								prev = { is_independent_ruler = yes }
							}
							trigger_event = { id = zzz_tribsub_subjugation.0001 }
						}
					}
					end_tributary = yes
				}
			}
		}
	}
}

# ---------- 0001 三选项效果(带叛逃者无效防御) ----------
# scope = 宗主(事件根)。若事件弹出时叛逃者已死/已臣服(tribsub_wayward_invalid),
# 选项只清理标记、不产生任何效果。
zzz_tribsub_revenge_choice_a_effect = {
	if = {
		limit = {
			NOT = { has_variable = tribsub_wayward_invalid }
		}
		# a. 立即兴师问罪
		add_prestige = -50
		# 记账入"报复列表"(支持多个叛逃者并存,不互相覆盖)
		add_to_variable_list = {
			name = tribsub_retaliation_target_list
			target = scope:wayward
		}
		# 叛逃者身上打账目标记(3 年过期,用于列表清理)
		scope:wayward = {
			set_variable = {
				name = tribsub_accounted
				value = yes
				days = 1095
			}
		}
		every_tributary = {
			limit = {
				has_subject_contract_group = tributary_subjugated
			}
			add_opinion = {
				modifier = tribsub_suzerain_aggression_opinion
				target = root
				opinion = -5
			}
		}
		if = {
			limit = {
				is_ai = yes
				is_at_war = no
			}
			ai_start_best_war = {
				cb = { tribsub_retaliation_cb }
				is_valid = {
					# 列表在宗主(root)身上,判断目标是否在列
					root = {
						is_target_in_variable_list = {
							name = tribsub_retaliation_target_list
							target = scope:target_character
						}
					}
					scope:target_character = {
						is_landed = yes
						is_independent_ruler = yes
					}
				}
			}
		}
	}
	remove_variable = tribsub_wayward_invalid
}

zzz_tribsub_revenge_choice_b_effect = {
	if = {
		limit = {
			NOT = { has_variable = tribsub_wayward_invalid }
		}
		# b. 认了
		add_prestige = -50
		every_tributary = {
			limit = {
				has_subject_contract_group = tributary_subjugated
			}
			set_variable = {
				name = tribsub_saw_suzerain_weakness
				value = yes
				days = 730
			}
		}
	}
	remove_variable = tribsub_wayward_invalid
}

zzz_tribsub_revenge_choice_c_effect = {
	if = {
		limit = {
			NOT = { has_variable = tribsub_wayward_invalid }
		}
		# c. 立威
		add_dread = 15
		add_prestige = -10
		scope:wayward = {
			add_opinion = {
				modifier = tribsub_resent_new_suzerain_opinion
				target = root
				opinion = -30
			}
		}
		set_relation_rival = {
			target = scope:wayward
			reason = tribsub_revenge_rival_reason
		}
		every_tributary = {
			limit = {
				has_subject_contract_group = tributary_subjugated
			}
			set_variable = {
				name = tribsub_fears_suzerain
				value = yes
				days = 730
			}
			add_opinion = {
				modifier = tribsub_intimidated_opinion
				target = root
				opinion = -10
			}
		}
	}
	remove_variable = tribsub_wayward_invalid
}
