# 被征服朝贡者:宗主继承后的反抗机制季度脉冲
# 挂载到官方 quarterly_playable_pulse(on_actions 追加,与原版合并,不覆盖)
# 每季度对每个有朝贡者的统治者执行:
#   - 宗主快照比对:换宗主 -> 打 2 年反抗期标记 + 好感惩罚 + 通知事件
#   - 反抗期 AI 每年掷一次反抗骰:强者宣独立战争,弱者直接断贡
#   - 宗主侧:断贡叛逃者复仇评估(AI 自动,无弹窗)
quarterly_playable_pulse = {
	on_actions = {
		zzz_tribsub_subjugated_tributary_pulse
		zzz_tribsub_suzerain_revenge_pulse
	}
}

zzz_tribsub_subjugated_tributary_pulse = {
	trigger = {
		number_of_tributaries > 0
	}

	effect = {
		every_tributary = {
			limit = {
				has_subject_contract_group = tributary_subjugated
			}
			zzz_tribsub_subjugated_tributary_check_effect = yes
		}
	}
}

# 宗主侧复仇评估(scope = 宗主,仅 AI,每年至多评估一次)
zzz_tribsub_suzerain_revenge_pulse = {
	trigger = {
		OR = {
			exists = var:tribsub_wayward_tributary
			has_variable_list = tribsub_retaliation_target_list
		}
	}

	effect = {
		# 账目清理:列表元素已重新成为朝贡者(收服成功)或账目标记已过期(3年)
		# 注意不清理"已开战"的情况——战争期间保留,否则 CB 合法性失效会无效化战争
		every_in_list = {
			variable = tribsub_retaliation_target_list
			if = {
				limit = {
					OR = {
						is_tributary = yes
						NOT = { has_variable = tribsub_accounted }
					}
				}
				root = {
					remove_list_variable = {
						name = tribsub_retaliation_target_list
						target = prev
					}
				}
			}
		}
		# wayward 单槽:目标已是朝贡者则清
		if = {
			limit = {
				var:tribsub_wayward_tributary = { is_tributary = yes }
			}
			remove_variable = tribsub_wayward_tributary
		}
		# L1:扫描"正在对我打独立战争"的被征服朝贡者 -> 记入报复列表
		# (覆盖:反抗掷骰宣战成功 / 玩家朝贡者主动宣战,两者都不经过断贡记账点)
		every_tributary = {
			limit = {
				has_subject_contract_group = tributary_subjugated
				is_at_war_with = root
			}
			add_to_variable_list = {
				name = tribsub_retaliation_target_list
				target = prev
			}
			set_variable = {
				name = tribsub_accounted
				value = yes
				days = 1095
			}
		}
		# AI 报复评估:只在打得过且未在交战时尝试重新收服
		if = {
			limit = {
				is_ai = yes
				is_at_war = no
				NOT = { exists = var:tribsub_wayward_tributary_cooldown }
			}
			ai_start_best_war = {
				cb = { tribsub_retaliation_cb }
				is_valid = {
					# 目标必须在报复列表(wayward 单槽断贡者由事件 a 入列,
					# 若事件未处理则走 L1 扫描,故此处只认列表)
					root = {
						is_target_in_variable_list = {
							name = tribsub_retaliation_target_list
							target = scope:target_character
						}
					}
					scope:target_character = {
						is_landed = yes
						in_diplomatic_range = root
						current_military_strength < root.current_military_strength
					}
				}
			}
			set_variable = {
				name = tribsub_wayward_tributary_cooldown
				value = yes
				days = 365
			}
		}
	}
}
