# 被征服朝贡者反抗机制事件
# 0001:宗主视角:朝贡者断贡叛逃 -> 三选一应对
# 0002:玩家是被征服朝贡者时,宗主死亡更替通知
# 0003:被征服朝贡者死亡后,继承人继承被征服朝贡关系时通知
namespace = zzz_tribsub_subjugation

zzz_tribsub_subjugation.0001 = {
	title = zzz_tribsub_subjugation.0001.t
	desc = zzz_tribsub_subjugation.0001.desc
	theme = war

	immediate = {
		var:tribsub_wayward_tributary = {
			save_scope_as = wayward
		}
		# 立即清账:防止玩家挂机期间新断贡者覆盖本事件对象
		remove_variable = tribsub_wayward_tributary
		# 防御:叛逃者已死或已臣服他人(触发前检查漏网)——
		# 打标记使所有选项只清理、不生效
		if = {
			limit = {
				OR = {
					NOT = { exists = scope:wayward }
					scope:wayward = { is_independent_ruler = no }
				}
			}
			set_variable = {
				name = tribsub_wayward_invalid
				value = yes
			}
		}
	}

	option = {
		name = zzz_tribsub_subjugation.0001.a
		zzz_tribsub_revenge_choice_a_effect = yes
		ai_chance = {
			base = 45
			modifier = {
				add = 10
				has_trait = brave
			}
			modifier = {
				add = 10
				OR = {
					has_trait = vengeful
					has_trait = wrathful
				}
			}
			modifier = {
				add = -25
				has_trait = craven
			}
			modifier = {
				add = 20
				current_military_strength > scope:wayward.current_military_strength
			}
		}
	}

	option = {
		name = zzz_tribsub_subjugation.0001.b
		zzz_tribsub_revenge_choice_b_effect = yes
		ai_chance = {
			base = 30
			modifier = {
				add = 15
				has_trait = craven
			}
			modifier = {
				add = 10
				OR = {
					has_trait = generous
					has_trait = forgiving
				}
			}
			modifier = {
				add = -10
				has_trait = brave
			}
		}
	}

	option = {
		name = zzz_tribsub_subjugation.0001.c
		zzz_tribsub_revenge_choice_c_effect = yes
		ai_chance = {
			base = 25
			modifier = {
				add = 10
				dread >= 50
			}
			modifier = {
				add = -10
				has_trait = craven
			}
			modifier = {
				add = -5
				has_trait = brave
			}
		}
	}
}

zzz_tribsub_subjugation.0002 = {
	title = zzz_tribsub_subjugation.0002.t
	desc = zzz_tribsub_subjugation.0002.desc
	theme = tributary

	option = {
		name = zzz_tribsub_subjugation.0002.a
	}

	option = {
		name = zzz_tribsub_subjugation.0002.b
		remove_variable = tribsub_resisting_new_suzerain
		# 宗主记账 + 弹出三选一应对事件(触发前防御:叛逃者仍独立才弹)
		suzerain = {
			set_variable = {
				name = tribsub_wayward_tributary
				value = prev
			}
			if = {
				limit = {
					prev = { is_independent_ruler = yes }
				}
				trigger_event = { id = zzz_tribsub_subjugation.0001 }
			}
		}
		end_tributary = yes
	}
}
