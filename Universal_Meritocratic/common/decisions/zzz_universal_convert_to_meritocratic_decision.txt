universal_convert_to_meritocratic_decision = {
	picture = {
		reference = "gfx/interface/illustrations/event_story/tgp_dynastic_cycle_stability.dds"
	} 
	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 60
		empire = 60
		hegemony = 0
	}

	desc = universal_convert_to_meritocratic_desc
	selection_tooltip = universal_convert_to_meritocratic_tooltip
	confirm_text = universal_meritocratic_confirm

	is_shown = {
		has_tgp_dlc_trigger = yes
		NOT = {
			government_has_flag = government_has_merit
		}
		NOR = {
			government_has_flag = government_is_theocracy
			government_has_flag = government_is_republic
			government_has_flag = government_is_holy_order
			government_has_flag = government_is_mercenary
			government_has_flag = government_is_nomadic
			government_has_flag = government_is_herder
		}
		any_sub_realm_county = {
			count > 0
			title_province = {
				OR = {
					has_holding_type = castle_holding
					has_holding_type = church_holding
					has_holding_type = city_holding
					has_holding_type = temple_citadel_holding
					has_holding_type = tribal_holding
				}
			}
		}
		is_playable_character = yes
	}

	is_valid = {
		trigger_if = {
			limit = {
				top_liege != this
			}
			custom_tooltip = {
				text = universal_meritocratic_liege_requirement
				liege = {
					government_has_flag = government_has_merit
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_at_war = no
		is_available_adult = yes
		has_active_diarchy = no
		is_tributary = no
	}

	cost = {
		gold = {
			value = 0
			if = {
				limit = {
					has_treasury = no
				}
				add = monumental_gold_value
				if = {
					limit = {
						highest_held_title_tier < tier_kingdom
					}
					multiply = 0.5
				}
			}
		}
		treasury = {
			value = 0
			if = {
				limit = {
					has_treasury = yes
				}
				add = monumental_gold_value
				if = {
					limit = {
						highest_held_title_tier < tier_kingdom
					}
					multiply = 0.5
				}
			}
		}
		prestige = {
			value = massive_prestige_value
			if = {
				limit = {
					highest_held_title_tier < tier_kingdom
				}
				multiply = 0.5
			}
		}
		influence = {
			value = monumental_influence_value
			if = {
				limit = {
					highest_held_title_tier < tier_kingdom
				}
				multiply = 0.5
			}
		}
	}

	effect = {
		add_legitimacy_effect = { LEGITIMACY = monumental_legitimacy_gain }
		dynasty = {
			add_dynasty_prestige = monumental_dynasty_prestige_gain
		}
		change_to_administrative_holding_effect = yes
		tgp_domicile_conversion_when_changing_government_type = {
			NEW_DOMICILE_TYPE = east_asian_estate
			NEW_GOVERNMENT_TYPE = meritocratic_government
		}
		change_government = meritocratic_government
		hidden_effect = {
			if = {
				limit = {
					OR = {
						has_treasury = no
						treasury = 0
					}
				}
				save_scope_as = new_treasury
			}
			if = {
				limit = {
					has_treasury = yes
					exists = scope:new_treasury
				}
				add_treasury = {
					value = gold
					min = 100
				}
			}
			if = {
				limit = {
					NOT = {
						any_held_title = {
							is_noble_family_title = yes
						}
					}
					exists = dynasty
				}
				create_noble_family_effect = { GOVERNMENT_GIVER = this }
			}
			every_tributary = {
				add_to_list = tributaries
			}
			every_in_list = {
				list = tributaries
				start_tributary_interaction_effect = {  # to update their contract types for your new status
					TRIBUTARY = this
					SUZERAIN = root
				}
			}
		}
		custom_tooltip = {
			text = universal_meritocratic_vassals_will_convert
			every_vassal_or_below = {
				limit = {
					NOT = {
						government_has_flag = government_has_merit
					}
					NOR = {
						government_has_flag = government_is_theocracy
						government_has_flag = government_is_republic
						government_has_flag = government_is_holy_order
						government_has_flag = government_is_mercenary
						government_has_flag = government_is_nomadic
						government_has_flag = government_is_herder
					}
				}
				change_to_administrative_effect = yes
			}
		}
		hidden_effect = {
			every_vassal_or_below = {
				limit = {
					government_has_flag = government_has_merit
					highest_held_title_tier >= tier_county
					NOT = {
						any_held_title = {
							is_noble_family_title = yes
						}
					}
					exists = dynasty
				}
				create_noble_family_effect = { GOVERNMENT_GIVER = top_liege }
			}
		}
	}

	ai_potential = {
	}

	ai_will_do = {
		base = 0
		modifier = {
			add = 20
			NOT = { has_trait = nomadic_philosophy }
		}
		modifier = {
			add = 30
			has_education_learning_trigger = yes
		}
		modifier = {
			add = 50
			has_religion = religion:confucianism_religion
		}
		modifier = {
			add = 30
			has_trait = confucian_education
		}
	}
}
