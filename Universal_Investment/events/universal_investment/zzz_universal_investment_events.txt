# ============================================================
# 0001 开工确认(活动到达后由 on_phase_active 触发,root=玩家)
# ============================================================
universal_investment.0001 = {
	type = character_event
	title = universal_investment.0001.t
	desc = universal_investment.0001.desc
	theme = stewardship
	override_background = {
		reference = terrain
	}

	left_portrait = {
		character = root
		animation = survey
	}

	option = {
		# 开工:扣阶梯建造成本,动工
		name = universal_investment.0001.a
		trigger = {
			location = {
				has_holding = no
				NOT = {
					has_ongoing_construction = yes
				}
			}
			gold >= universal_investment_construction_cost_value
		}
		custom_tooltip = universal_investment.0001.a.tt
		universal_investment_start_construction_effect = yes
		ai_chance = {
			base = 100
		}
	}

	option = {
		# 反悔:不扣建造成本,仅亏差旅费
		name = universal_investment.0001.b
		ai_chance = {
			base = 0
		}
	}
}

# ============================================================
# 0005 年金年度处理(隐藏,由 yearly_playable_pulse 触发,玩家与 AI 共用)
# ============================================================
universal_investment.0005 = {
	type = character_event
	hidden = yes

	trigger = {
		is_alive = yes
		is_landed = yes
		has_variable_list = universal_investment_portfolio
	}

	immediate = {
		root = {
			set_variable = {
				name = universal_investment_report_gold
				value = 0
			}
			set_variable = {
				name = universal_investment_report_loss
				value = 0
			}
		}
		every_in_list = {
			variable = universal_investment_portfolio
			universal_investment_process_annuity_effect = yes
		}
		if = {
			limit = {
				is_ai = no
				OR = {
					var:universal_investment_report_gold >= 1
					var:universal_investment_report_loss >= 1
				}
			}
			trigger_event = {
				id = universal_investment.0007
			}
		}
	}
}

# ============================================================
# 0011 建造监控(隐藏,月度自循环,root=玩家)
# 逐笔检查在建投资:完工结算 / 取消停跟 / 风险判定
# 无 pending 后链自灭(watcher_active 复位,下次开工重开)
# ============================================================
universal_investment.0011 = {
	type = character_event
	hidden = yes

	trigger = {
		is_alive = yes
		has_variable_list = universal_investment_pending
	}

	immediate = {
		every_in_list = {
			variable = universal_investment_pending
			universal_investment_check_pending_effect = yes
		}
		if = {
			limit = {
				any_in_list = {
					variable = universal_investment_pending
					var:universal_investment_pending = yes
				}
			}
			trigger_event = {
				id = universal_investment.0011
				months = 1
			}
		}
		else = {
			remove_variable = universal_investment_watcher_active
		}
	}
}
